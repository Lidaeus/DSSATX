```powershell
python -m src.reporting.export_posterior_demo_data `
  --run-id t3_1_s4_20260718 `
  --posterior-dir workspace/runs/t3_1_s4_20260718/t4_posterior_demo_c `
  --candidate-dir workspace/runs/t3_1_s4_20260718/t4_candidate_predictions_v2 `
  --out-dir workspace/runs/t3_1_s4_20260718/t4_paper_data_demo `

& 'C:\Program Files\R\R-4.5.2\bin\Rscript.exe' paper/R/make_all.R `
  --input-dir workspace/runs/t3_1_s4_20260718/t4_paper_data_demo `
  --out-dir workspace/runs/t3_1_s4_20260718/t4_paper_figures_demo `
  --figure-ids figs5_posterior,figs6_spearman,figs7_equifinality `
  --allow-unvalidated-demo
```
